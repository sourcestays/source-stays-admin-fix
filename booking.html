-- Source Stays Credit: secure, admin-managed rewards ledger -- Run once in Supabase SQL Editor. Review any existing schema differences before running. ALTER TABLE public.bookings ADD COLUMN IF NOT EXISTS stay_completed_at timestamptz, ADD COLUMN IF NOT EXISTS source_stays_credit_applied numeric(12,2) NOT NULL DEFAULT 0 CHECK (source_stays_credit_applied >= 0); CREATE TABLE IF NOT EXISTS public.source_stays_credit_ledger ( id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, customer_phone text NOT NULL, amount numeric(12,2) NOT NULL CHECK (amount <> 0), entry_type text NOT NULL CHECK (entry_type IN ('earn', 'redeem', 'adjustment')), booking_id bigint REFERENCES public.bookings(id) ON DELETE RESTRICT, note text NOT NULL DEFAULT '', created_by uuid NOT NULL DEFAULT auth.uid(), created_at timestamptz NOT NULL DEFAULT now() ); CREATE INDEX IF NOT EXISTS source_stays_credit_phone_created_idx ON public.source_stays_credit_ledger (customer_phone, created_at DESC); -- One earning entry per completed booking, and one redemption per booking. CREATE UNIQUE INDEX IF NOT EXISTS source_stays_credit_one_earn_per_booking ON public.source_stays_credit_ledger (booking_id) WHERE entry_type = 'earn' AND booking_id IS NOT NULL; CREATE UNIQUE INDEX IF NOT EXISTS source_stays_credit_one_redemption_per_booking ON public.source_stays_credit_ledger (booking_id) WHERE entry_type = 'redeem' AND booking_id IS NOT NULL; ALTER TABLE public.source_stays_credit_ledger ENABLE ROW LEVEL SECURITY; -- No direct table access for anonymous/public users. Admins use the RPC functions below. REVOKE ALL ON public.source_stays_credit_ledger FROM anon, authenticated; DROP POLICY IF EXISTS "Source Stays admins can read rewards ledger" ON public.source_stays_credit_ledger; CREATE POLICY "Source Stays admins can read rewards ledger" ON public.source_stays_credit_ledger FOR SELECT TO authenticated USING (EXISTS (SELECT 1 FROM public.admin_users au WHERE au.id = auth.uid())); DROP POLICY IF EXISTS "Source Stays admins can insert rewards ledger" ON public.source_stays_credit_ledger; CREATE OR REPLACE FUNCTION public.source_stays_credit_balance(p_customer_phone text) RETURNS numeric LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_balance numeric(12,2); BEGIN IF auth.uid() IS NULL OR NOT EXISTS ( SELECT 1 FROM public.admin_users au WHERE au.id = auth.uid() ) THEN RAISE EXCEPTION 'Only Source Stays admins can view rewards balances'; END IF; SELECT COALESCE(SUM(amount), 0)::numeric(12,2) INTO v_balance FROM public.source_stays_credit_ledger WHERE customer_phone = regexp_replace(COALESCE(p_customer_phone, ''), '[^0-9]', '', 'g'); RETURN v_balance; END; $$; CREATE OR REPLACE FUNCTION public.source_stays_credit_earn(p_booking_id bigint) RETURNS numeric LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_booking public.bookings%ROWTYPE; v_price numeric(12,2); v_nights integer; v_eligible numeric(12,2); v_reward numeric(12,2); v_phone text; BEGIN IF auth.uid() IS NULL OR NOT EXISTS ( SELECT 1 FROM public.admin_users au WHERE au.id = auth.uid() ) THEN RAISE EXCEPTION 'Only Source Stays admins can issue rewards'; END IF; SELECT * INTO v_booking FROM public.bookings WHERE id = p_booking_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Booking not found'; END IF; IF lower(COALESCE(v_booking.status, '')) <> 'approved' THEN RAISE EXCEPTION 'Booking must be approved before rewards can be issued'; END IF; IF lower(COALESCE(v_booking.payment_status, '')) <> 'paid' THEN RAISE EXCEPTION 'Booking must be marked paid before rewards can be issued'; END IF; IF v_booking.stay_completed_at IS NULL THEN RAISE EXCEPTION 'Mark the stay completed before issuing rewards'; END IF; v_phone := regexp_replace(COALESCE(v_booking.customer_phone, ''), '[^0-9]', '', 'g'); IF v_phone = '' THEN RAISE EXCEPTION 'Booking has no customer phone number'; END IF; SELECT COALESCE(p.price, 0)::numeric(12,2) INTO v_price FROM public.properties p WHERE p.id = v_booking.property_id; IF v_price IS NULL OR v_price <= 0 THEN RAISE EXCEPTION 'Property accommodation price is missing or invalid'; END IF; v_nights := (v_booking.check_out::date - v_booking.check_in::date); IF v_nights <= 0 THEN RAISE EXCEPTION 'Booking dates are invalid'; END IF; -- 2% applies only to accommodation price x nights, excluding deposits and fees. v_eligible := v_price * v_nights; v_reward := round(v_eligible * 0.02, 2); INSERT INTO public.source_stays_credit_ledger (customer_phone, amount, entry_type, booking_id, note, created_by) VALUES (v_phone, v_reward, 'earn', p_booking_id, '2% Source Stays Credit for completed paid stay; eligible accommodation: NGN ' || v_eligible, auth.uid()); RETURN v_reward; END; $$; CREATE OR REPLACE FUNCTION public.source_stays_credit_redeem( p_booking_id bigint, p_customer_phone text, p_amount numeric ) RETURNS numeric LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE v_booking public.bookings%ROWTYPE; v_phone text; v_balance numeric(12,2); v_amount numeric(12,2); v_remaining_due numeric(12,2); BEGIN IF auth.uid() IS NULL OR NOT EXISTS ( SELECT 1 FROM public.admin_users au WHERE au.id = auth.uid() ) THEN RAISE EXCEPTION 'Only Source Stays admins can redeem rewards'; END IF; v_phone := regexp_replace(COALESCE(p_customer_phone, ''), '[^0-9]', '', 'g'); v_amount := round(COALESCE(p_amount, 0), 2); IF v_phone = '' THEN RAISE EXCEPTION 'Customer phone is required'; END IF; IF v_amount <= 0 THEN RAISE EXCEPTION 'Redemption amount must be greater than zero'; END IF; SELECT * INTO v_booking FROM public.bookings WHERE id = p_booking_id FOR UPDATE; IF NOT FOUND THEN RAISE EXCEPTION 'Booking not found'; END IF; IF regexp_replace(COALESCE(v_booking.customer_phone, ''), '[^0-9]', '', 'g') <> v_phone THEN RAISE EXCEPTION 'Customer phone does not match this booking'; END IF; IF v_booking.check_in::date <= current_date THEN RAISE EXCEPTION 'Credit can only be applied to a future booking'; END IF; IF lower(COALESCE(v_booking.status, '')) = 'declined' THEN RAISE EXCEPTION 'Cannot redeem credit on a cancelled booking'; END IF; -- Serialize all redemptions for this normalized customer phone, including when no ledger rows exist yet. PERFORM pg_advisory_xact_lock(hashtextextended(v_phone, 0)); SELECT COALESCE(SUM(amount), 0)::numeric(12,2) INTO v_balance FROM public.source_stays_credit_ledger WHERE customer_phone = v_phone; IF v_amount > v_balance THEN RAISE EXCEPTION 'Requested redemption exceeds available credit balance'; END IF; v_remaining_due := GREATEST(COALESCE(v_booking.amount_due, 0) - COALESCE(v_booking.source_stays_credit_applied, 0), 0); IF v_amount > v_remaining_due THEN RAISE EXCEPTION 'Redemption cannot exceed the remaining booking amount due'; END IF; INSERT INTO public.source_stays_credit_ledger (customer_phone, amount, entry_type, booking_id, note, created_by) VALUES (v_phone, -v_amount, 'redeem', p_booking_id, 'Source Stays Credit applied to future booking', auth.uid()); UPDATE public.bookings SET source_stays_credit_applied = COALESCE(source_stays_credit_applied, 0) + v_amount WHERE id = p_booking_id; RETURN v_balance - v_amount; END; $$; REVOKE ALL ON FUNCTION public.source_stays_credit_balance(text) FROM PUBLIC, anon; REVOKE ALL ON FUNCTION public.source_stays_credit_earn(bigint) FROM PUBLIC, anon; REVOKE ALL ON FUNCTION public.source_stays_credit_redeem(bigint, text, numeric) FROM PUBLIC, anon; GRANT EXECUTE ON FUNCTION public.source_stays_credit_balance(text) TO authenticated; GRANT EXECUTE ON FUNCTION public.source_stays_credit_earn(bigint) TO authenticated; GRANT EXECUTE ON FUNCTION public.source_stays_credit_redeem(bigint, text, numeric) TO authenticated; COMMENT ON TABLE public.source_stays_credit_ledger IS 'Admin-managed Source Stays Credit ledger. Positive amounts earn credit; negative amounts redeem credit.'; COMMENT ON COLUMN public.bookings.stay_completed_at IS 'Set by Source Stays admin only after the guest stay has actually been completed.'; COMMENT ON COLUMN public.bookings.source_stays_credit_applied IS 'Source Stays Credit applied to this booking; payment flow must subtract this from the amount due.';
